# Python packages that are installed on the Pi Zero globally using apt
# DO NOT install these on the Pi Zero using pip
# These do NOT go in the .venv
# If already installed, the command just verifies that, or upgrades it.
sudo apt install python3-numpy      # Python test: import numpy as np
sudo apt install python3-rpi-lgpio  # Python test: import RPi.GPIO as GPIO (legacy?)
sudo apt install python3-gpiozero   # Python test:
sudo apt install python3-opencv     # Python test: import cv2
sudo apt install python3-h5py       # Python test: import h5py
sudo apt install python3-yaml       # Python test: import yaml

# python3-h5py: capture archives (src/hdf5write.py, src/capture_h5.py).
#   Never pip install h5py here — it builds from source against the HDF5
#   headers, which on a Zero takes hours and usually exhausts memory. The
#   apt package is prebuilt against the system libhdf5 and pulls it in.
# python3-yaml: reads config/instrument.yaml (src/config_manager.py).
#   Usually already present on Raspberry Pi OS.
